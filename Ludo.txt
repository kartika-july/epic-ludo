            function finalizeRoll() {
                const color = players[currentPlayerIndex];
                let rollResult = Math.floor(Math.random() * 6) + 1;
                const allInBase = pawns[color].every(p => p.step === -1);
                
                // Sistem Anti-Sial & Dadu Hoki Ultimate
                if (forcedDice === 6 && color === 'green') {
                    rollResult = 6; forcedDice = 0;
                } else if (rollResult !== 6) {
                    consecutiveNoSix[color]++;
                    if (allInBase && consecutiveNoSix[color] >= 4) rollResult = 6;
                }
                if (rollResult === 6) consecutiveNoSix[color] = 0; 
                
                currentDiceValue = rollResult;
                diceElement.innerText = diceFaces[currentDiceValue - 1];
                diceElement.style.transform = 'rotateX(0deg) rotateY(0deg) scale(1)';
                isRolling = false; hasRolled = true;
                addLog(`🎲 ${playerNames[color]} mendapat angka <b>${currentDiceValue}</b>`, color);
                checkMovablePawns();
            }

                    else if (pawn.step <= 55) { if (pawn.step + currentDiceValue <= 56) canMove = true; }

            function handlePawnClick(color, id, pawnEl) {
                if (!hasRolled || isMoving || color !== players[currentPlayerIndex] || !pawnEl.classList.contains('pawn-selectable')) return;
                document.querySelectorAll('.pawn-selectable').forEach(p => p.classList.remove('pawn-selectable'));

                    pawn.step = 0; renderPawns(); updateScore(color, 10, "Keluar Markas", pawnEl);
                    setTimeout(() => finishMove(color, pawn, pawnEl), 300);
                } else {
                    let stepsLeft = currentDiceValue;
                    const moveInterval = setInterval(() => {
                        pawn.step++; renderPawns(); SFX.move(); stepsLeft--;
                        if (stepsLeft <= 0) { clearInterval(moveInterval); setTimeout(() => finishMove(color, pawn, pawnEl), 300); }
                    }, 250);
                }
            }

            function finishMove(color, movedPawn, pawnEl, skipMystery = false) {
                let extraTurn = (currentDiceValue === 6);

                if (movedPawn.step === 56) { updateScore(color, 100, "Bidak Finish!", pawnEl); SFX.win(); }

                if (movedPawn.step >= 0 && movedPawn.step <= 50) {
                    const currentCoords = getPawnCoordinates(color, movedPawn.step);
                    const mainIndex = (playerConfig[color].startIndex + movedPawn.step) % 52;
                    
                    // Cek Kotak Misteri Dulu
                    if (!skipMystery && mysteryIndices.includes(mainIndex)) {
                        triggerMysteryBox(color, movedPawn, pawnEl);
                        return; // Hentikan alur normal, pindah ke fungsi kotak misteri
                    }

                    const isSafe = safeIndices.includes(mainIndex);
                    if (isSafe && movedPawn.step !== 0) {
                        updateScore(color, 20, "Safe Zone", pawnEl);
                    } else if (!isSafe) {
                        players.forEach(otherColor => {
                            if (otherColor !== color) {
                                let enemies = [];
                                pawns[otherColor].forEach((otherPawn, oIdx) => {
                                    if (otherPawn.step >= 0 && otherPawn.step <= 50) {
                                        const otherCoords = getPawnCoordinates(otherColor, otherPawn.step);
                                        if (otherCoords && otherCoords.r === currentCoords.r && otherCoords.c === currentCoords.c) {
                                            enemies.push({ pawn: otherPawn, idx: oIdx });
                                        }
                                    }
                                });

                                if (enemies.length === 1) {
                                    // Cek Kubah Es
                                    if (otherColor === 'blue' && activeShields.blue > 0) {
                                        addLog(`❄️ Serangan Gagal! Bidak Biru dilindungi Kubah Es!`, 'blue');
                                        SFX.magic();
                                    } else {
                                        enemies[0].pawn.step = -1; extraTurn = true; SFX.capture();
                                        updateScore(color, 50, `Memakan lawan`, pawnEl);
                                        const eatenEl = pawnElements[otherColor][enemies[0].idx];
                                        createExplosion(eatenEl, otherColor);
                                        eatenEl.style.transform = 'scale(0)';
                                        setTimeout(() => { eatenEl.style.transform = 'scale(1)'; renderPawns(); }, 300);
                                    }
                                } else if (enemies.length > 1) {
                                    addLog(`🛡️ Terbentur benteng musuh!`, otherColor); SFX.move();
                                }
                            }
                        });
                    }
                }

                renderPawns();

                if (pawns[color].every(p => p.step === 56)) {
                    updateScore(color, 500, "MENANG!", null); triggerWin(color); return;
                }

                if (extraTurn) {
                    hasRolled = false; isMoving = false; btnRoll.classList.remove('opacity-50', 'cursor-not-allowed');
                    turnStatus.innerText = "Bonus Kocok!"; gameStatus.innerText = "Dapat bonus! Kocok lagi.";
                } else { setTimeout(nextTurn, 500); }
            }

                    pawn.step += direction; if (pawn.step < 0) pawn.step = -1;
                    renderPawns(); SFX.move(); stepCount--;
                    if (stepCount <= 0 || pawn.step === -1 || pawn.step === 56) {
                        clearInterval(interval); finishMove(color, pawn, pawnEl, true); // skipMystery = true
                    }
                }, 200);
            }
